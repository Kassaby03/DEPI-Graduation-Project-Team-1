EXEC sys.sp_cdc_enable_db;

-- ???? CDC ??? ?? ???? ?? schema ???? (????: dbo)
DECLARE @sql NVARCHAR(MAX) = N'';
SELECT @sql = @sql + 'EXEC sys.sp_cdc_enable_table @source_schema = ''' + s.name + ''', @source_name = ''' + t.name + ''', @role_name = NULL; '
FROM sys.tables t
JOIN sys.schemas s ON t.schema_id = s.schema_id
WHERE s.name = 'dbo';

EXEC sp_executesql @sql;
